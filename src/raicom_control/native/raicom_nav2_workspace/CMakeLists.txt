cmake_minimum_required(VERSION 3.16)
project(raicom_nav2_workspace)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_lifecycle nav2_core nav2_costmap_2d nav2_util dwb_core dwb_msgs nav_msgs nav_2d_msgs pluginlib tf2 tf2_geometry_msgs tf2_ros std_msgs geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(raicom_nav2_workspace SHARED src/workspace_controller.cpp)
target_compile_features(raicom_nav2_workspace PUBLIC cxx_std_17)
target_compile_options(raicom_nav2_workspace PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(raicom_nav2_workspace PRIVATE include)
ament_target_dependencies(raicom_nav2_workspace rclcpp rclcpp_lifecycle nav2_core nav2_costmap_2d nav2_util dwb_core dwb_msgs nav_msgs nav_2d_msgs pluginlib tf2 tf2_geometry_msgs tf2_ros std_msgs geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
pluginlib_export_plugin_description_file(dwb_core plugins.xml)
add_library(raicom_body_clear_layer SHARED src/body_clear_layer.cpp)
target_compile_features(raicom_body_clear_layer PUBLIC cxx_std_17)
target_compile_options(raicom_body_clear_layer PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(raicom_body_clear_layer PRIVATE include)
ament_target_dependencies(raicom_body_clear_layer nav2_costmap_2d pluginlib)
pluginlib_export_plugin_description_file(nav2_costmap_2d costmap_plugins.xml)
install(TARGETS raicom_body_clear_layer LIBRARY DESTINATION lib)
install(TARGETS raicom_nav2_workspace LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(approach_policy_test test/approach_policy_test.cpp)
  target_compile_features(approach_policy_test PRIVATE cxx_std_17)
  target_compile_options(approach_policy_test PRIVATE -UNDEBUG)
  target_include_directories(approach_policy_test PRIVATE include)
  add_test(NAME approach_policy COMMAND approach_policy_test)
  add_executable(body_clear_layer_test test/body_clear_layer_test.cpp)
  target_compile_features(body_clear_layer_test PRIVATE cxx_std_17)
  target_compile_options(body_clear_layer_test PRIVATE -UNDEBUG)
  target_include_directories(body_clear_layer_test PRIVATE include)
  ament_target_dependencies(body_clear_layer_test nav2_costmap_2d)
  add_test(NAME body_clear_layer COMMAND body_clear_layer_test)
endif()
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
